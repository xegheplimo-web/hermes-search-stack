A) gateway/providers/_http.py:
   - FetchFn = Callable[..., str]  # fetch(url, *, method="GET", headers=None, params=None, json=None, timeout=None) -> response TEXT
   - def default_fetch(url, *, method="GET", headers=None, params=None, json=None, timeout=30.0) -> str:
       lazy httpx import (raise ProviderError-like RuntimeError with a clear message when missing);
       returns response text (utf-8, errors="replace"); non-2xx -> RuntimeError(f"HTTP {status} from {url}: {body[:200]}").
   - def parse_mcp_body(body: str) -> str — copied from standalone (JSON or SSE "data:" lines; error envelopes raise).
   - def mcp_call(fetch: FetchFn, url: str, tool: str, arguments: dict) -> str — POST JSON-RPC tools/call
     with headers {"Content-Type": "application/json", "Accept": "application/json, text/event-stream"},
     then parse_mcp_body on the response text.
   - def parse_exa_search_text(text: str, limit: int) -> list[SearchItem] — copied from standalone (--- separated
     Title:/URL:/Highlights: blocks).

B) exa.py — class ExaProvider: name="exa", capabilities=frozenset({"search", "extract"})
   __init__(self, fetch: FetchFn | None = None) -> self._fetch = fetch or default_fetch
   EXA_MCP_URL = "https://mcp.exa.ai/mcp"
   search(query, max_results=10): mcp_call(self._fetch, EXA_MCP_URL, "web_search_exa",
     {"query": query, "numResults": max(1, int(max_results))}) -> parse_exa_search_text(text, max_results).
   extract(urls, char_limit=15000): per URL mcp_call(..., "web_fetch_exa", {"urls": [url]}) ->
     ExtractItem(url=url, title=<first "# "/"Title:" line>, content=text[:char_limit]).

C) parallel.py — class ParallelProvider: name="parallel", capabilities={"search", "extract"}
   PARALLEL_MCP_URL = "https://search.parallel.ai/mcp"
   search(query, max_results=10): mcp_call(..., "web_search", {"objective": query, "search_queries": [query],
     "session_id": uuid.uuid4().hex}) -> json.loads(text) -> results[] ->
     SearchItem(title=r["title"], url=r["url"], description=" ".join(r.get("excerpts") or []), position=i+1).
   extract(urls, char_limit=15000): per URL mcp_call(..., "web_fetch", {"urls": [url], "objective": "Full page content",
     "session_id": ...}) -> json: results[] -> full_content/content/excerpts join; errors[] -> ExtractItem(url, error=...).

D) jina.py — class JinaProvider: name="jina", capabilities=frozenset({"extract"})  (extract-ONLY)
   JINA_READER = "https://r.jina.ai/"
   extract(urls, char_limit=15000): per URL: self._fetch(JINA_READER + url, headers={"Accept": "text/plain"})
     -> parse: title from the "Title:" line; content = text AFTER the "Markdown Content:" marker (whole text if absent).
   search(...): raise ProviderError("jina", "search not supported (extract-only provider)")
     (lazy import: `from gateway.providers import ProviderError` INSIDE the method — avoids circular import).
   CRITICAL (verified live): do NOT send a browser User-Agent — Cloudflare challenges it; `Accept: text/plain` is REQUIRED.

E) v2ex.py — class V2EXProvider: name="v2ex", capabilities={"search"}
   search(query, max_results=10): self._fetch("https://www.sov2ex.com/api/search",
     params={"q": query, "size": max(1, int(max_results))}) -> json: hits[]._source ->
     SearchItem(title=src["title"], url=f"https://www.v2ex.com/t/{src['id']}",
     description=(src.get("content") or "")[:200], position=i+1).

F) bilibili.py — class BilibiliProvider: name="bilibili", capabilities={"search"}
   BROWSER_UA = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36"
   search(query, max_results=10): self._fetch("https://api.bilibili.com/x/web-interface/search/all/v2",
     params={"keyword": query, "page": 1}, headers={"User-Agent": BROWSER_UA}) -> json: data.result[] ->
     the group with result_type=="video" -> data[] items: title = strip "<em>"/"</em>" tags + html.unescape,
     url = "https://www.bilibili.com/video/" + (bvid or from arcurl), description = desc or author, position=i+1.

G) youtube.py — class YouTubeProvider: name="youtube", capabilities={"search"}  (pure HTTP, NO yt-dlp)
   search(query, max_results=10): self._fetch("https://www.youtube.com/results",
     params={"search_query": query}, headers={"User-Agent": BROWSER_UA}) -> regex
     `var ytInitialData = (\{.*?\});</script>` (DOTALL) -> json.loads -> walk the tree for dicts with key
     "videoRenderer" -> SearchItem(title=r["title"]["runs"][0]["text"], url=f"https://www.youtube.com/watch?v={r['videoId']}",
     description = first of shortBylineText runs / lengthText / "" , position=i+1). Cap at max_results.

H) rss.py — class RSSProvider: name="rss", capabilities={"search"}  (in-repo parser, NO feedparser)

     channel/item -> SearchItem(title=<title>, url=<link>, description = source text + pubDate best-effort, position=i+1).
